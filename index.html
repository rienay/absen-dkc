Panduan Setup Google Sheets (3 Menit)

  1. Buka Google Sheets dan buat Spreadsheet baru (misal: "Database Presensi DKC Cilacap").
  2. Klik menu Ekstensi (Extensions) → Apps Script.
  3. Hapus kode bawaan di editor, lalu klik tombol "Salin Kode Script" di bawah ini dan tempelkan.
  4. Klik ikon Simpan.
  5. Klik tombol Terapkan (Deploy) → Penerapan baru (New deployment).
  6. Pilih jenis Aplikasi Web (Web App).
  7. Atur Jalankan sebagai: Saya (Me) dan Siapa yang memiliki akses: Siapa saja (Anyone).
  8. Klik Terapkan, setujui izin akun Google, lalu salin URL Aplikasi Web (Web App URL).
  9. Tempelkan URL tersebut ke menu Pengaturan di aplikasi ini dan klik Simpan & Hubungkan.
Kode Google Apps Script: 
const SHEET_MEMBERS = "Anggota";
const SHEET_ATTENDANCE = "Presensi";

function initSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheetMembers = ss.getSheetByName(SHEET_MEMBERS);
  if (!sheetMembers) {
    sheetMembers = ss.insertSheet(SHEET_MEMBERS);
    const headers = ["ID", "NTA", "Nama", "Unit", "Status", "Pangkalan", "Jabatan", "Golongan", "Kontak", "Catatan", "Dibuat_Pada"];
    sheetMembers.appendRow(headers);
    sheetMembers.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#0f3d27").setFontColor("#ffffff");
    sheetMembers.setFrozenRows(1);
  }
  let sheetAtt = ss.getSheetByName(SHEET_ATTENDANCE);
  if (!sheetAtt) {
    sheetAtt = ss.insertSheet(SHEET_ATTENDANCE);
    const headers = ["ID", "MemberId", "Tanggal", "NTA", "Nama", "Unit", "Aktivitas", "Selfie_URL", "Lokasi", "Waktu_Presensi"];
    sheetAtt.appendRow(headers);
    sheetAtt.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#0f3d27").setFontColor("#ffffff");
    sheetAtt.setFrozenRows(1);
  }
}

function doGet(e) {
  try {
    initSheets();
    const action = e && e.parameter ? e.parameter.action : "getAll";
    if (action === "ping") {
      return jsonResponse({ status: "success", message: "Google Sheets API Presensi DKC Cilacap Aktif!", timestamp: new Date().toISOString() });
    }
    return jsonResponse({ status: "success", data: getAllData(), timestamp: new Date().toISOString() });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function doPost(e) {
  try {
    initSheets();
    let body = {};
    if (e && e.postData && e.postData.contents) {
      body = JSON.parse(e.postData.contents);
    } else if (e && e.parameter) {
      body = e.parameter;
    }
    const action = body.action;
    if (action === "saveMember") {
      return jsonResponse({ status: "success", data: saveMember(body.member) });
    }
    if (action === "deleteMember") {
      deleteMember(body.id);
      return jsonResponse({ status: "success", message: "Anggota dihapus" });
    }
    if (action === "saveAttendance") {
      return jsonResponse({ status: "success", data: saveAttendance(body.attendance) });
    }
    if (action === "deleteAttendance") {
      deleteAttendance(body.id);
      return jsonResponse({ status: "success", message: "Presensi dihapus" });
    }
    return jsonResponse({ status: "error", message: "Action POST tidak valid" });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function getAllData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const mSheet = ss.getSheetByName(SHEET_MEMBERS);
  const aSheet = ss.getSheetByName(SHEET_ATTENDANCE);
  const members = [];
  if (mSheet && mSheet.getLastRow() > 1) {
    const rows = mSheet.getRange(2, 1, mSheet.getLastRow() - 1, 11).getValues();
    rows.forEach(r => {
      if (r[0]) members.push({ id: String(r[0]), nta: String(r[1]||""), nama: String(r[2]||""), unit: String(r[3]||""), status: String(r[4]||"Aktif Tugas"), pangkalan: String(r[5]||""), jabatan: String(r[6]||""), golongan: String(r[7]||""), kontak: String(r[8]||""), catatan: String(r[9]||""), createdAt: r[10] ? String(r[10]) : "" });
    });
  }
  const attendance = [];
  if (aSheet && aSheet.getLastRow() > 1) {
    const rows = aSheet.getRange(2, 1, aSheet.getLastRow() - 1, 10).getValues();
    rows.forEach(r => {
      if (r[0]) attendance.push({ id: String(r[0]), memberId: String(r[1]||""), date: String(r[2]||""), nta: String(r[3]||""), nama: String(r[4]||""), unit: String(r[5]||""), activity: String(r[6]||""), selfie: String(r[7]||""), location: String(r[8]||""), time: r[9] ? String(r[9]) : "" });
    });
  }
  return { members, attendance };
}

function saveMember(member) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_MEMBERS);
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0]));
    const index = ids.indexOf(String(member.id));
    if (index !== -1) {
      sheet.getRange(index + 2, 1, 1, 10).setValues([[member.id, member.nta, member.nama, member.unit, member.status, member.pangkalan, member.jabatan || "-", member.golongan, member.kontak || "", member.catatan || ""]]);
      return member;
    }
  }
  sheet.appendRow([member.id, member.nta, member.nama, member.unit, member.status, member.pangkalan, member.jabatan || "-", member.golongan, member.kontak || "", member.catatan || "", new Date().toLocaleString("id-ID", { timeZone: "Asia/Jakarta" })]);
  return member;
}

function deleteMember(id) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_MEMBERS);
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;
  const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0]));
  const index = ids.indexOf(String(id));
  if (index !== -1) sheet.deleteRow(index + 2);
}

function saveAttendance(att) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_ATTENDANCE);
  const lastRow = sheet.getLastRow();
  let selfieRef = att.selfie || "";
  if (lastRow > 1) {
    const data = sheet.getRange(2, 1, lastRow - 1, 3).getValues();
    for (let i = 0; i < data.length; i++) {
      if (String(data[i][0]) === String(att.id) || (String(data[i][1]) === String(att.memberId) && String(data[i][2]) === String(att.date))) {
        sheet.getRange(i + 2, 1, 1, 9).setValues([[att.id, att.memberId, att.date, att.nta, att.nama, att.unit, att.activity, selfieRef, att.location || "-"]]);
        return att;
      }
    }
  }
  sheet.appendRow([att.id, att.memberId, att.date, att.nta, att.nama, att.unit, att.activity, selfieRef, att.location || "-", new Date().toLocaleString("id-ID", { timeZone: "Asia/Jakarta" })]);
  return att;
}

function deleteAttendance(id) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_ATTENDANCE);
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;
  const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues().map(r => String(r[0]));
  const index = ids.indexOf(String(id));
  if (index !== -1) sheet.deleteRow(index + 2);
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
